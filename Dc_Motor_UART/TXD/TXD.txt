#include<reg51.h>
#include<uart.h>
void timer_delay(int);
char tx_data,count=0;
bit flag=0;
void ISR_Speed(void)interrupt 0
{
		if(count<5)
		count++;
		if(count==5)
		 count=1;
		tx_data=count+'0';
		
}
void ISR_Direct(void)interrupt 2
{
	flag=~flag;
	if(flag==0)
	{
		tx_data='A';
	}
	else
	{
		tx_data='B';
	}
}
void main()
{
	UART_Init();
	EA=1;
	EX0=1;
	EX1=1;
	IT0=1;
	IT1=1;
	while(1)
	{
		UART_Tx(tx_data);
	}
}
